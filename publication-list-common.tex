\newbibmacro*{publication-identifier}{%
  \printtext{\textbf{Identifier:}\space}%
  \iffieldundef{doi}
    {\iffieldundef{isbn}
       {\iffieldundef{issn}
          {\iffieldundef{eprint}
             {\iffieldundef{url}
                {\textcolor{missingid}{TO BE COMPLETED}}
                {URL\addcolon\space
                 \href{\thefield{url}}{\nolinkurl{\thefield{url}}}}}
             {\thefield{eprinttype}\addcolon\space\thefield{eprint}}}
          {ISSN\addcolon\space\thefield{issn}}}
       {ISBN\addcolon\space\thefield{isbn}}}
    {DOI\addcolon\space\href{https://doi.org/\thefield{doi}}{\nolinkurl{\thefield{doi}}}}}

\newbibmacro*{candidate-contribution}{%
  \par\smallskip
  \printtext{\textbf{Candidate's contribution:}\space}%
  \ifcsdef{publicationcontribution@\thefield{entrykey}}
    {\csuse{publicationcontribution@\thefield{entrykey}}}
    {\DefaultCandidateContribution}}

\renewbibmacro*{finentry}{%
  % Discard the standard driver's pending delimiter before starting the
  % administrative metadata on a fresh line.
  \setunit{}%
  \par\smallskip\usebibmacro{publication-identifier}%
  \usebibmacro{candidate-contribution}%
  \finentry}
